cmake_minimum_required(VERSION 3.21)
project(netft_viewer VERSION 0.1.1 LANGUAGES CXX)
file(
  STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/core/netft/UPSTREAM"
  NETFT_VIEWER_CORE_COMMIT_LINES
  REGEX "^commit="
)
list(LENGTH NETFT_VIEWER_CORE_COMMIT_LINES NETFT_VIEWER_CORE_COMMIT_COUNT)
if(NOT NETFT_VIEWER_CORE_COMMIT_COUNT EQUAL 1)
  message(FATAL_ERROR "core/netft/UPSTREAM must contain exactly one commit entry")
endif()
list(GET NETFT_VIEWER_CORE_COMMIT_LINES 0 NETFT_VIEWER_CANONICAL_CORE_SNAPSHOT)
string(REGEX REPLACE "^commit=" "" NETFT_VIEWER_CANONICAL_CORE_SNAPSHOT
  "${NETFT_VIEWER_CANONICAL_CORE_SNAPSHOT}")
set(
  NETFT_VIEWER_CORE_SNAPSHOT
  "${NETFT_VIEWER_CANONICAL_CORE_SNAPSHOT}"
  CACHE INTERNAL
  "netft-cpp snapshot identity embedded in the companion"
  FORCE
)
string(LENGTH "${NETFT_VIEWER_CORE_SNAPSHOT}" NETFT_VIEWER_CORE_SNAPSHOT_LENGTH)
if(
  NOT NETFT_VIEWER_CORE_SNAPSHOT_LENGTH EQUAL 40
  OR NOT NETFT_VIEWER_CORE_SNAPSHOT MATCHES "^[0-9a-f]+$"
)
  message(FATAL_ERROR "NETFT_VIEWER_CORE_SNAPSHOT must be a 40-character commit ID")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_subdirectory(core)

option(
  NETFT_VIEWER_BUILD_HARDWARE_TEST
  "Build the opt-in physical sensor verification executable"
  OFF
)
if(NETFT_VIEWER_BUILD_HARDWARE_TEST)
  add_subdirectory(test/hardware)
endif()
